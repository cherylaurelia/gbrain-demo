=== ./src/server.js
import express from 'express';
import { listsForHousehold } from './lists.js';

const app = express();
app.use(express.json());

app.get('/households/:id/lists', async (req, res) => {
  res.json(await listsForHousehold(req.params.id));
});

app.post('/checkout', async (req, res) => {
  const { householdId, plan } = req.body ?? {};
  if (!householdId || !plan) {
    return res.status(400).json({ error: 'householdId and plan are required' });
  }
  try {
    const r = await fetch(`${process.env.PAYMENT_API_URL}/checkouts`, {
      method: 'POST',
      headers: {
        'content-type': 'application/json',
        authorization: `Bearer ${process.env.PAYMENT_API_KEY}`,
        'idempotency-key': req.get('idempotency-key') ?? '',
      },
      body: JSON.stringify({ reference: householdId, plan }),
    });
    if (!r.ok) return res.status(502).json({ error: 'payment provider error' });
    res.status(201).json(await r.json());
  } catch {
    res.status(502).json({ error: 'payment provider unreachable' });
  }
});

app.listen(process.env.PORT || 3000);
=== ./src/lists.js
import pg from 'pg';
const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL });

export async function listsForHousehold(id) {
  const { rows } = await pool.query('select * from lists where household_id = $1', [id]);
  return rows;
}
