=== ./src/server.js
import express from 'express';
import { Paddle } from '@paddle/paddle-node-sdk';
import { listsForHousehold } from './lists.js';

const paddle = new Paddle(process.env.PADDLE_API_KEY);
const app = express();
app.use(express.json());

app.get('/households/:id/lists', async (req, res) => {
  res.json(await listsForHousehold(req.params.id));
});

app.post('/checkout', async (req, res) => {
  const { householdId } = req.body ?? {};
  if (!householdId) return res.status(400).json({ error: 'householdId required' });
  try {
    const txn = await paddle.transactions.create({
      items: [{ priceId: process.env.PADDLE_PRICE_ID, quantity: 1 }],
      customData: { householdId },
    });
    res.status(201).json({ checkoutUrl: txn.checkout?.url, transactionId: txn.id });
  } catch (err) {
    console.error('paddle checkout failed', err);
    res.status(502).json({ error: 'payment provider error' });
  }
});

app.listen(process.env.PORT || 3000);
=== ./src/lists.js
import pg from 'pg';
const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL });

export async function listsForHousehold(id) {
  const { rows } = await pool.query('select * from lists where household_id = $1', [id]);
  return rows;
}
