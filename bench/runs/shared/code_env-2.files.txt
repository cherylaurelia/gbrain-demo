=== ./src/server.js
import express from 'express';
import { listsForHousehold } from './lists.js';

const app = express();
app.use(express.json());

app.get('/households/:id/lists', async (req, res) => {
  res.json(await listsForHousehold(req.params.id));
});

app.listen(process.env.PORT || 3000);
=== ./src/lists.js
import pg from 'pg';
const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL });

export async function listsForHousehold(id) {
  const { rows } = await pool.query('select * from lists where household_id = $1', [id]);
  return rows;
}
