=== ./src/server.js
import express from 'express';
import { listsForHousehold } from './lists.js';
import { createCheckout } from './checkout.js';

const app = express();
app.use(express.json());

app.get('/households/:id/lists', async (req, res) => {
  res.json(await listsForHousehold(req.params.id));
});

app.post('/checkout', async (req, res) => {
  const { householdId, priceId } = req.body ?? {};
  if (!householdId || !priceId) {
    return res.status(400).json({ error: 'householdId and priceId are required' });
  }
  try {
    res.status(201).json(await createCheckout(householdId, priceId));
  } catch (err) {
    console.error('checkout failed', err);
    res.status(502).json({ error: 'payment provider error' });
  }
});

app.listen(process.env.PORT || 3000);
=== ./src/lists.js
import pg from 'pg';
const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL });

export async function listsForHousehold(id) {
  const { rows } = await pool.query('select * from lists where household_id = $1', [id]);
  return rows;
}
=== ./src/checkout.js
import { Paddle, Environment } from '@paddle/paddle-node-sdk';

const paddle = new Paddle(process.env.PADDLE_API_KEY ?? '', {
  environment: process.env.PADDLE_ENV === 'production' ? Environment.production : Environment.sandbox,
});

// Creates a Paddle transaction for the household's subscription price and
// returns the hosted checkout URL.
export async function createCheckout(householdId, priceId) {
  const txn = await paddle.transactions.create({
    items: [{ priceId, quantity: 1 }],
    customData: { household_id: householdId },
  });
  return { id: txn.id, url: txn.checkout?.url ?? null };
}
